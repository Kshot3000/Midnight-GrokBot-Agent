    Teaching Compact sketch

    Illustrative sealed-message pattern — not a compilable contract. Verify helpers against Compact docs before any real work.

    pragma language_version >= 0.23;

import CompactStandardLibrary;

/* Public: thread id + message commitment. Private body is a witness. */
export ledger threadCommit: Bytes<32>;
export ledger msgCount: Counter;

witness sealedBody(): Bytes<32>;

export circuit postSealed(threadTag: Bytes<32>): [] {
  /* Teaching stand-in — verify persistentHash / pad before compile. */
  const body = sealedBody();
  threadCommit = persistentHash<Bytes<32>>(
    pad(32, "nocturne:msg:") || threadTag || body
  );
  msgCount.increment(1);
  disclose(threadCommit);
  disclose(msgCount);
  /* Body never becomes ledger. */
}
