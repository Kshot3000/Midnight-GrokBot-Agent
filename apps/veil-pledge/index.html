Veil Pledge Studio

World-class educational tip jar for Midnight’s dual-state model: seal a private amount + note, commit only a public hash to the simulated ledger, then prove a threshold (amount ≥ X) without disclosing the exact tip — or selectively disclose when you choose. By @kshot9000 · NightDream.io.

    ✗ Cleartext tip on ledger
    

    export ledger tipAda: Uint<64>;

export circuit tip(amount: Uint<64>): [] {
  // Everyone sees the tip forever
  tipAda = amount;
}
    
        
    
        
    
          
    ✓ Commitment + selective disclose
    

    witness tipAmount(): Uint<64>;
witness tipSalt(): Bytes<32>;

export ledger commitment: Bytes<32>;

export circuit seal(): [] {
  commitment = disclose(
    persistentHash<…>([tipAmount(), tipSalt()])
  );
}

// Later: prove tipAmount() >= threshold in-circuit
// without writing the exact amount to the ledger.
